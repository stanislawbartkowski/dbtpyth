# dbt-core 2.0.0a4 on PyPI ships dbt-fusion (a compiled binary), which
# doesn't support method=session or Python models on Spark -- both of
# which this project's Python model needs. Pin to a classic (pure Python)
# 1.x release instead.
dbt-core==1.10.22
dbt-spark[session]==1.10.1

# dbt-core/dbt-adapters pin mashumaro<3.15/<3.18, but that version fails
# to import on Python 3.14 (mashumaro.exceptions.UnserializableField on
# JSONObjectSchema). Force a newer, 3.14-compatible mashumaro; this
# violates dbt-core's declared upper bound (harmless in practice), so
# install with: pip install -r requirements.txt --use-deprecated=legacy-resolver
mashumaro==3.22
